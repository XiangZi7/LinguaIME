cmake_minimum_required(VERSION 3.24)
project(LinguaIME LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The TSF host requires Windows")
endif()
set(LINGUA_RUST_DIR "${CMAKE_SOURCE_DIR}/../../target/release" CACHE PATH "ime-ffi DLL and import library")
add_library(lingua_tsf SHARED src/dll.cpp src/registration.cpp src/text_service.cpp src/candidate_window.cpp lingua_tsf.def)
target_include_directories(lingua_tsf PRIVATE include)
target_compile_definitions(lingua_tsf PRIVATE UNICODE _UNICODE)
target_compile_options(lingua_tsf PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(lingua_tsf PRIVATE "${LINGUA_RUST_DIR}/ime_ffi.dll.lib" ole32 oleaut32 uuid advapi32 user32 gdi32 dwmapi)
add_executable(native_contract_test tests/key_policy_test.cpp)
target_include_directories(native_contract_test PRIVATE include)
target_link_libraries(native_contract_test PRIVATE "${LINGUA_RUST_DIR}/ime_ffi.dll.lib")
enable_testing()
add_test(NAME native_contract COMMAND native_contract_test)
