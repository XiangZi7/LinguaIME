cmake_minimum_required(VERSION 3.24)
project(LinguaIME LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The TSF host requires Windows")
endif()
set(LINGUA_RUST_DIR "${CMAKE_SOURCE_DIR}/../../target/release" CACHE PATH "ime-ffi DLL and import library")
set(LINGUA_RIME_DIR "${CMAKE_SOURCE_DIR}/../../build/native/rime" CACHE PATH "Prepared offline Rime bundle")
if(NOT EXISTS "${LINGUA_RIME_DIR}/runtime-id.txt")
  message(FATAL_ERROR "Prepare the Rime bundle with scripts/prepare-rime.ps1 before building the native host")
endif()
set(INPUT_SOURCES src/text_service.cpp src/input_mode.cpp src/language_bar.cpp src/candidate_window.cpp src/candidate_renderer.cpp)
set(LOADER_SOURCES src/bootstrap_service.cpp src/runtime_loader.cpp)
add_library(lingua_tsf SHARED src/dll.cpp src/registration.cpp ${LOADER_SOURCES} lingua_tsf.def)
target_compile_definitions(lingua_tsf PRIVATE LINGUA_BOOTSTRAP)
target_link_libraries(lingua_tsf PRIVATE ole32 oleaut32 uuid advapi32 user32)
target_link_options(lingua_tsf PRIVATE /Brepro)
add_library(lingua_runtime SHARED src/dll.cpp ${INPUT_SOURCES} lingua_runtime.def)
target_link_libraries(lingua_runtime PRIVATE "${LINGUA_RUST_DIR}/ime_ffi.dll.lib" ole32 oleaut32 uuid advapi32 user32 gdi32 dwmapi delayimp d2d1 dwrite)
target_link_options(lingua_runtime PRIVATE /Brepro /DELAYLOAD:ime_ffi.dll /DELAY:UNLOAD)
target_compile_options(lingua_runtime PRIVATE /O2)
add_custom_command(TARGET lingua_runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LINGUA_RUST_DIR}/ime_ffi.dll" "$<TARGET_FILE_DIR:lingua_runtime>")
add_custom_command(TARGET lingua_runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${LINGUA_RIME_DIR}" "$<TARGET_FILE_DIR:lingua_runtime>/rime")
add_executable(native_contract_test tests/key_policy_test.cpp)
target_link_libraries(native_contract_test PRIVATE "${LINGUA_RUST_DIR}/ime_ffi.dll.lib")
add_executable(tsf_context_test tests/tsf_context_test.cpp ${INPUT_SOURCES})
target_compile_definitions(tsf_context_test PRIVATE LINGUA_TESTING)
target_link_libraries(tsf_context_test PRIVATE "${LINGUA_RUST_DIR}/ime_ffi.dll.lib" ole32 oleaut32 uuid advapi32 user32 gdi32 dwmapi d2d1 dwrite)
add_executable(profile_tool tests/profile_tool.cpp)
target_link_libraries(profile_tool PRIVATE ole32 oleaut32 uuid user32)
add_executable(hot_reload_test tests/hot_reload_test.cpp ${LOADER_SOURCES})
target_compile_definitions(hot_reload_test PRIVATE LINGUA_TESTING)
target_link_libraries(hot_reload_test PRIVATE ole32 oleaut32 uuid user32)
foreach(flavor IN ITEMS a b)
  add_library(mock_runtime_${flavor} SHARED tests/mock_runtime.cpp tests/mock_runtime.def)
  set_target_properties(mock_runtime_${flavor} PROPERTIES OUTPUT_NAME "mock-runtime-${flavor}")
  target_link_libraries(mock_runtime_${flavor} PRIVATE ole32 oleaut32 uuid)
endforeach()
target_compile_definitions(mock_runtime_a PRIVATE MOCK_FLAVOR=1)
target_compile_definitions(mock_runtime_b PRIVATE MOCK_FLAVOR=2)
foreach(native_target IN ITEMS lingua_tsf lingua_runtime native_contract_test tsf_context_test profile_tool hot_reload_test mock_runtime_a mock_runtime_b)
  target_include_directories(${native_target} PRIVATE include)
  target_compile_definitions(${native_target} PRIVATE UNICODE _UNICODE)
  target_compile_options(${native_target} PRIVATE /W4 /WX /EHsc /utf-8)
endforeach()
enable_testing()
add_test(NAME native_contract COMMAND native_contract_test)
add_test(NAME tsf_context COMMAND tsf_context_test)
add_test(NAME hot_reload COMMAND hot_reload_test)
set_tests_properties(native_contract tsf_context hot_reload PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:hot_reload_test>" ENVIRONMENT "LINGUAIME_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-data")
